cmake_minimum_required(VERSION 3.16)
project(BNSim VERSION 3.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()
set(BNSIM_SANITIZER "" CACHE STRING "Sanitizer: address (includes undefined), thread, or undefined")
if(BNSIM_SANITIZER)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Sanitizers require GCC or Clang")
  endif()
  if(BNSIM_SANITIZER STREQUAL "address")
    set(BNSIM_SAN_FLAGS -fsanitize=address,undefined)
  elseif(BNSIM_SANITIZER STREQUAL "thread")
    set(BNSIM_SAN_FLAGS -fsanitize=thread)
  elseif(BNSIM_SANITIZER STREQUAL "undefined")
    set(BNSIM_SAN_FLAGS -fsanitize=undefined)
  else()
    message(FATAL_ERROR "Unknown BNSIM_SANITIZER")
  endif()
  add_compile_options(${BNSIM_SAN_FLAGS} -fno-omit-frame-pointer -g)
  add_link_options(${BNSIM_SAN_FLAGS})
endif()
add_library(bnsim src/model.cpp src/motility.cpp src/simulation.cpp src/transport.cpp src/biology.cpp src/mechanics.cpp src/io.cpp)
target_include_directories(bnsim PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_link_libraries(bnsim PUBLIC Threads::Threads)
execute_process(COMMAND git describe --always --dirty WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} OUTPUT_VARIABLE BNSIM_VERSION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT BNSIM_VERSION)
  set(BNSIM_VERSION "source-archive")
endif()
file(GLOB_RECURSE BNSIM_SOURCE_FILES CONFIGURE_DEPENDS "src/*.cpp" "src/*.hpp" "include/*.hpp")
list(APPEND BNSIM_SOURCE_FILES "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${BNSIM_SOURCE_FILES})
set(BNSIM_HASH_INPUT "")
foreach(source IN LISTS BNSIM_SOURCE_FILES)
  file(SHA256 "${source}" source_hash)
  string(APPEND BNSIM_HASH_INPUT "${source_hash}")
endforeach()
string(SHA256 BNSIM_SOURCE_HASH "${BNSIM_HASH_INPUT}")
target_compile_definitions(bnsim PRIVATE BNSIM_CODE_VERSION="${BNSIM_VERSION}" BNSIM_SOURCE_SHA256="${BNSIM_SOURCE_HASH}")
target_compile_definitions(bnsim PRIVATE BNSIM_COMPILER="${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}" BNSIM_BUILD_TYPE="${CMAKE_BUILD_TYPE}")
add_executable(chemotaxis_example src/apps/example.cpp)
target_link_libraries(chemotaxis_example PRIVATE bnsim)
add_executable(qs_biofilm_example src/apps/example.cpp)
target_compile_definitions(qs_biofilm_example PRIVATE BNSIM_QS_EXAMPLE=1)
target_link_libraries(qs_biofilm_example PRIVATE bnsim)
include(CTest)
if(BUILD_TESTING)
  add_executable(bnsim_tests tests/test_main.cpp)
  target_link_libraries(bnsim_tests PRIVATE bnsim)
  target_include_directories(bnsim_tests PRIVATE src)
  target_compile_definitions(bnsim_tests PRIVATE BNSIM_TEST_DATA="${CMAKE_CURRENT_SOURCE_DIR}/tests/reference")
  foreach(group IN ITEMS validation transport biology mechanics motility reproducibility)
    add_test(NAME ${group} COMMAND bnsim_tests ${group})
  endforeach()
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME examples_and_outputs COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_examples.py $<TARGET_FILE:chemotaxis_example> $<TARGET_FILE:qs_biofilm_example>)
endif()
